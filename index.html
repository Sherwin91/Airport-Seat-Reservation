// Producer thread function: Passenger booking generator
void* producer_thread(void* arg) {
sem_wait(&empty); // Wait for available buffer slot
pthread_mutex_lock(&buffer_mutex); // Acquire buffer lock
insert_request_to_buffer(req);
pthread_mutex_unlock(&buffer_mutex);
sem_post(&full); // Signal full slot available
}

// Consumer thread function: Booking agent
void* consumer_agent(void* arg) {
sem_wait(&full); // Wait for requests in buffer
pthread_mutex_lock(&buffer_mutex);
BookingRequest req = remove_request_from_buffer();
pthread_mutex_unlock(&buffer_mutex);
sem_post(&empty); // Signal empty slot freed

// Critical Section: Flight Seat Database access
pthread_mutex_lock(&seat_mutex); // Lock seat database
if (seat[req.seat].status == AVAILABLE) {
seat[req.seat].status = RESERVED;
}
pthread_mutex_unlock(&seat_mutex); // Unlock seat database
}
📚 OS Practical Viva Voce Cheatsheet
Top oral examination questions
Q1: What is the Producer-Consumer Problem?
It is a classic multi-process synchronization problem where one set of threads (producers) generates items into a bounded shared queue, and another set (consumers) removes them, ensuring no buffer overflow or underflow occurs.
Q2: Why do we need both Semaphores and Mutexes?
Counting semaphores (empty and full) track available/occupied buffer slot counts and block threads when the queue is full or empty. The Mutex (buffer_mutex / seat_mutex) guarantees Mutual Exclusion so only one thread modifies shared queue data structures or seat states at any instant.
Q3: What causes a Race Condition in seat reservation?
When two concurrent booking agents check and write a seat state without a mutex lock. Both see the seat as AVAILABLE, and both attempt to write RESERVED, leading to data corruption and duplicate seat booking.
Q4: What is a Critical Section?
The section of code where shared resources (e.g. flight seat array or buffer pointers) are accessed and updated. Access must be mutually exclusive.
